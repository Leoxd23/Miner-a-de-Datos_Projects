from pathlib import Path
import numpy as np
import pandas as pd
import matplotlib.pyplot as plt

# Buscar automáticamente el archivo de referencia en las ubicaciones disponibles.
rutas_csv = [
    Path(r"C:\Users\leoen\Documents\MINERIA DE DATOS\Practica 1\dataset_limpio.csv"),
    Path(r"C:\Users\leoen\Downloads\Practica 1\dataset_limpio.csv"),
]
ruta_datos = next((ruta for ruta in rutas_csv if ruta.exists() and ruta.stat().st_size > 0), None)

if ruta_datos is not None:
    df = pd.read_csv(ruta_datos)
    fuente = str(ruta_datos)
else:
    # Fallback reproducible para poder ejecutar la práctica aunque el CSV esté vacío.
    rng = np.random.default_rng(42)
    n = 240
    recipe_names = [
        "Creamy Garlic Parmesan Pasta", "Banana Bread", "Tomato Soup",
        "Chicken Tacos", "Greek Salad", "Brownies", "Fried Rice",
        "Oatmeal Cookies"
    ]
    df = pd.DataFrame({
        "recipe_name": rng.choice(recipe_names, n),
        "stars": np.clip(np.rint(rng.normal(4.1, 0.8, n)), 1, 5),
        "thumbs_up": rng.poisson(28, n),
        "thumbs_down": rng.poisson(4, n),
        "reply_count": rng.poisson(3, n),
        "best_score": rng.integers(20, 101, n),
    })
    fuente = "Reproducible demonstration data (reference CSV has no records)"

# Normalizar columnas numéricas y eliminar filas que no permitan graficar.
columnas_numericas = ["stars", "thumbs_up", "thumbs_down", "reply_count", "best_score"]
for columna in columnas_numericas:
    if columna in df.columns:
        df[columna] = pd.to_numeric(df[columna], errors="coerce")

columnas_faltantes = set(columnas_numericas) - set(df.columns)
if "recipe_name" not in df.columns or columnas_faltantes:
    raise ValueError(f"Missing required columns: {sorted(columnas_faltantes | ({'recipe_name'} - set(df.columns)))}")

df = df.dropna(subset=columnas_numericas + ["recipe_name"]).copy()
print(f"Source: {fuente}")
print(f"Records analyzed: {len(df):,}")

# Preparar resúmenes una sola vez para reutilizarlos en las seis gráficas.
resumen_recetas = (
    df.groupby("recipe_name", as_index=False)
    .agg(
        resenas=("recipe_name", "size"),
        promedio_estrellas=("stars", "mean"),
        me_gusta=("thumbs_up", "sum"),
        no_me_gusta=("thumbs_down", "sum"),
    )
    .sort_values("resenas", ascending=False)
)
top_recetas = resumen_recetas.head(8).sort_values("resenas")

plt.style.use("seaborn-v0_8-whitegrid")
fig, axes = plt.subplots(3, 2, figsize=(16, 18))
fig.suptitle("Automated visualization of recipe reviews", fontsize=20, fontweight="bold", y=0.995)

# Cada elemento del ciclo define una gráfica distinta y su configuración.
graficas = [
    ("Bar chart", lambda ax: ax.barh(top_recetas["recipe_name"], top_recetas["resenas"], color="#2563eb")),
    ("Pie chart", lambda ax: ax.pie(top_recetas["resenas"], labels=top_recetas["recipe_name"], autopct="%1.1f%%", startangle=90, textprops={"fontsize": 8})),
    ("Histogram", lambda ax: ax.hist(df["stars"], bins=np.arange(0.5, 5.6, 1), color="#f59e0b", edgecolor="white", rwidth=0.85)),
    ("Box plot", lambda ax: ax.boxplot([df.loc[df["recipe_name"] == recipe_name, "stars"] for recipe_name in top_recetas["recipe_name"]], tick_labels=top_recetas["recipe_name"], patch_artist=True, boxprops={"facecolor": "#10b981", "alpha": 0.65})),
    ("Scatter plot", lambda ax: ax.scatter(df["thumbs_up"], df["stars"], c=df["best_score"], cmap="viridis", alpha=0.65, edgecolors="none")),
    ("Line chart", lambda ax: ax.plot(df.groupby("stars")["best_score"].mean().index, df.groupby("stars")["best_score"].mean().values, marker="o", color="#dc2626", linewidth=2)),
]

for ax, (chart_type, draw_chart) in zip(axes.flat, graficas):
    draw_chart(ax)
    ax.set_title(chart_type, fontsize=14, fontweight="bold")
    ax.tick_params(axis="x", rotation=35, labelsize=8)
    ax.tick_params(axis="y", labelsize=8)
    ax.grid(alpha=0.25)

axes[0, 0].set_xlabel("resenas")
axes[0, 1].set_ylabel("")
axes[1, 0].set_xlabel("stars")
axes[1, 0].set_ylabel("frequency")
axes[1, 1].set_xlabel("recipe_name")
axes[1, 1].set_ylabel("stars")
axes[2, 0].set_xlabel("thumbs_up")
axes[2, 0].set_ylabel("stars")
axes[2, 1].set_xlabel("stars")
axes[2, 1].set_ylabel("mean best_score")
fig.text(0.5, 0.01, "Six chart types generated with a configuration list and a for loop.", ha="center", fontsize=10)
plt.tight_layout(rect=[0, 0.03, 1, 0.97])
plt.show()

print("Generated chart types: bar, pie, histogram, box, scatter, and line.")
